-- Create database
CREATE DATABASE ecommerce_db;

-- Select database
USE ecommerce_db;


-- Create products table
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    category VARCHAR(50),
    brand VARCHAR(50),
    price DECIMAL(10,2),
    quantity INT,
    city VARCHAR(50),
    status VARCHAR(20)
);

-- Display table structure
DESC products;

-- Display available tables
SHOW TABLES;
